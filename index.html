const threshold = 60; function selectScenario(scenario) { let description; let risk; if (scenario === "safe") { description = "The road ahead is clear."; risk = 10; } else if (scenario === "traffic") { description = "A slower vehicle is travelling ahead."; risk = 45; } else if (scenario === "hazard") { description = "The vehicle ahead has suddenly braked."; risk = 90; } document.getElementById("environment").textContent = description; document.getElementById("risk").textContent = "Risk Score: " + risk + "%"; evaluateRisk(risk); } function evaluateRisk(risk) { const warning = document.getElementById("warning"); const status = document.getElementById("status"); if (risk >= threshold) { warning.textContent = "COLLISION WARNING"; warning.style.color = "#ef4444"; status.textContent = "Status: Potential collision detected"; } else { warning.textContent = "NO WARNING"; warning.style.color = "#4ade80"; status.textContent = "Status: Monitoring Road"; } }
